import React, { useState, useEffect } from 'react';
import { useAuth } from '@/contexts/AuthContext';
import PayButton from '@/components/PayButton';

export function CloudBaseTest() {
  const { user, subscription, loading, signIn, signOut } = useAuth();
  const [testResult, setTestResult] = useState('');

  const handleAnonymousLogin = async () => {
    try {
      await signIn('anonymous');
      setTestResult('✅ 匿名登录成功！');
    } catch (error) {
      setTestResult(`❌ 登录失败: ${error.message}`);
    }
  };

  return (
    <div style={{ padding: '2rem', border: '1px solid #ccc', borderRadius: '8px', marginTop: '2rem' }}>
      <h2>🧪 CloudBase 功能测试</h2>

      {/* 状态显示 */}
      <div style={{ marginBottom: '1rem', padding: '1rem', background: '#f5f5f5', borderRadius: '4px' }}>
        <h3>当前状态</h3>
        {loading ? (
          <p>⏳ 加载中...</p>
        ) : (
          <>
            <p><strong>用户:</strong> {user ? `${user.uid} (${user.loginType})` : '未登录'}</p>
            <p><strong>订阅计划:</strong> {subscription?.plan || 'N/A'}</p>
            <p><strong>订阅状态:</strong> {subscription?.status || 'N/A'}</p>
            {subscription?.expiresAt && (
              <p><strong>过期时间:</strong> {new Date(subscription.expiresAt).toLocaleDateString()}</p>
            )}
          </>
        )}
      </div>

      {/* 测试按钮 */}
      <div style={{ marginBottom: '1rem' }}>
        <h3>功能测试</h3>
        {!user ? (
          <button
            onClick={handleAnonymousLogin}
            style={{
              padding: '0.75rem 1.5rem',
              fontSize: '1rem',
              backgroundColor: '#3498db',
              color: 'white',
              border: 'none',
              borderRadius: '4px',
              cursor: 'pointer',
              marginRight: '1rem'
            }}
          >
            🔐 匿名登录测试
          </button>
        ) : (
          <>
            <button
              onClick={signOut}
              style={{
                padding: '0.75rem 1.5rem',
                fontSize: '1rem',
                backgroundColor: '#e74c3c',
                color: 'white',
                border: 'none',
                borderRadius: '4px',
                cursor: 'pointer',
                marginRight: '1rem'
              }}
            >
              🚪 登出
            </button>

            {subscription?.plan === 'free' && (
              <PayButton plan="member" text="🎁 测试开通会员" />
            )}
          </>
        )}
      </div>

      {/* 测试结果 */}
      {testResult && (
        <div style={{ padding: '1rem', background: testResult.includes('✅') ? '#d4edda' : '#f8d7da', borderRadius: '4px' }}>
          <strong>测试结果:</strong> {testResult}
        </div>
      )}

      {/* 说明 */}
      <div style={{ marginTop: '2rem', fontSize: '0.9rem', color: '#666' }}>
        <h4>测试说明</h4>
        <ol>
          <li><strong>匿名登录测试</strong>: 点击按钮测试匿名登录功能</li>
          <li><strong>订阅查询</strong>: 登录后自动查询订阅状态</li>
          <li><strong>会员开通测试</strong>: 登录后测试开通会员功能（开发模式，无需真实支付）</li>
          <li><strong>登出测试</strong>: 测试登出功能</li>
        </ol>
        <p><strong>注意</strong>: 当前为开发测试模式，所有功能都使用本地模拟数据。</p>
      </div>
    </div>
  );
}
